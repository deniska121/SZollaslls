AddCSLuaFile("shared.lua")
AddCSLuaFile("cl_init.lua")
include("shared.lua")

function ENT:Initialize()
	self:PhysicsInit(SOLID_VPHYSICS)
	self:SetMoveType(MOVETYPE_VPHYSICS)
	self:SetSolid(SOLID_VPHYSICS)
	self.itime = CurTime()
	local phys = self:GetPhysicsObject()
	if IsValid(phys) then phys:Wake() end
end

function ENT:PhysicsCollide(data)
	if self.TrowType ~= "molotov" or self.Exploded or data.Speed < 50 then return end
	self.Exploded = true
	if self:WaterLevel() >= 2 then self:Remove() return end
	self:EmitSound("physics/glass/glass_bottle_break1.wav")
	local attacker = IsValid(self:GetOwner()) and self:GetOwner() or self
	if CreateVFireBall then
		for i = 1, 7 do
			local angle = (i - 1) * math.pi * 2 / 7
			CreateVFireBall(25, 80, data.HitPos + data.HitNormal * 8, Vector(math.cos(angle) * 100, math.sin(angle) * 100, 70), attacker)
		end
	end
	for _, ent in ipairs(ents.FindInSphere(data.HitPos, 160)) do
		if ent == self or not IsValid(ent) then continue end
		local tr = util.TraceLine({start = data.HitPos + data.HitNormal * 2, endpos = ent:WorldSpaceCenter(), filter = {self, ent}, mask = MASK_SOLID_BRUSHONLY})
		if tr.Hit then continue end
		if IsValid(ent) then ent:Ignite(10, 0, attacker) end
	end
	timer.Simple(0, function() if IsValid(self) then self:Remove() end end)
end
