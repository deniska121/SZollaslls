function McINV.GetWeaponName(class)
    if class == "bandage" then return "Бинт" end
    return WeaponNames[class] or class
end

Config = Config or {}
Config.MaxWeight = 8
Config.ItemWeights = Config.ItemWeights or {
    ["bandage"] = 0.2,
    ["lockpick"] = 0.1,
}

for k, v in pairs(McINV.picktime_class) do
    Config.ItemWeights[k] = math.Clamp(v * 0.5, 1, 5)
end

function McINV.GetItemWeight(itemClass)
    return Config.ItemWeights[itemClass] or 0.5
end

function McINV.GetTotalWeight(ply)
    if not IsValid(ply) then return 0 end
    if not ply.Inventory then return 0 end
    local total = 0
    for _, item in ipairs(ply.Inventory) do
        total = total + McINV.GetItemWeight(item.class or item)
    end
    return math.Round(total, 2)
end

function McINV.CanAddItem(ply, itemClass)
    if not IsValid(ply) then return false end
    local current = McINV.GetTotalWeight(ply)
    local weight = McINV.GetItemWeight(itemClass)
    return (current + weight) <= Config.MaxWeight
end

local gunstates = {
    { "Отличное", 0.9, Color(0, 255, 0) },
    { "Хорошее", 0.7, Color(100, 255, 100) },
    { "Нормальное", 0.5, Color(255, 255, 0) },
    { "Потёртое", 0.3, Color(255, 150, 0) },
    { "Убитое", 0.1, Color(255, 0, 0) },
}

function McINV.GetGunState(gunstate)
    if not gunstate then return "Отличное", Color(0, 255, 0) end
    for _, v in ipairs(gunstates) do
        if gunstate >= v[2] then
            return v[1], v[3]
        end
    end
    return "Убитое", Color(255, 0, 0)
end

    function McINV.AddItem(ply, itemClass, weaponData)
        if not IsValid(ply) then return false end
        if not McINV.CanAddItem(ply, itemClass) then
            return false
        end
        ply.Inventory = ply.Inventory or {}
        local item = { class = itemClass }
        if weaponData then
            if weaponData.clip1 then item.clip1 = weaponData.clip1 end
            if weaponData.clip2 then item.clip2 = weaponData.clip2 end
            if weaponData.gunstate then item.gunstate = weaponData.gunstate end
            if weaponData.att_saved then item.att_saved = weaponData.att_saved end
            if weaponData.attachments then item.attachments = weaponData.attachments end
            if weaponData.modifications then item.modifications = weaponData.modifications end
            if weaponData.ammo_type then item.ammo_type = weaponData.ammo_type end
            if weaponData.primary_ammo then item.primary_ammo = weaponData.primary_ammo end
            if weaponData.secondary_ammo then item.secondary_ammo = weaponData.secondary_ammo end
            if weaponData.weapon_data then item.weapon_data = weaponData.weapon_data end
        end
        table.insert(ply.Inventory, item)
        McINV.ForceUpdate(ply)
        McINV.RefreshUI(ply)
        McINV.ForceInventoryPage(ply)
        return true
    end

    function McINV.RemoveItem(ply, index)
        if not IsValid(ply) then return false end
        ply.Inventory = ply.Inventory or {}
        if ply.Inventory[index] then
            table.remove(ply.Inventory, index)
            McINV.ForceUpdate(ply)
            McINV.RefreshUI(ply)
            McINV.ForceInventoryPage(ply)
            return true
        end
        return false
    end

    function McINV.SpawnWeaponBag(ply, item, spawnPos)
        local isHeavy = McINV.IsHeavyWeapon(item.class)
        local bagModel = isHeavy and "models/cityrp/bags/bag_drop.mdl" or "models/cityrp/bags/satchel_drop.mdl"
        
        local bag = ents.Create("prop_physics")
        if not IsValid(bag) then return false end
        
        bag:SetModel(bagModel)
        bag:SetPos(spawnPos)
        bag:SetAngles(Angle(0, math.random(0, 360), 0))
        bag:Spawn()
        bag:Activate()
        
        bag.NY_BagWeapon = item.class
        bag.NY_BagClip1 = item.clip1 or 0
        bag.NY_BagClip2 = item.clip2 or 0
        bag.NY_BagGunstate = item.gunstate or 1
        bag.NY_BagAttachments = item.att_saved or {}
        bag.NY_BagMods = item.modifications or {}
        bag.NY_BagAmmoType = item.ammo_type or ""
        bag.NY_BagPrimaryAmmo = item.primary_ammo or 0
        bag.NY_BagSecondaryAmmo = item.secondary_ammo or 0
        bag.NY_BagWeaponData = item.weapon_data or {}
        bag.NY_IsWeaponBag = true
        bag.NY_IsHeavy = isHeavy
        
        local phys = bag:GetPhysicsObject()
        if IsValid(phys) then
            phys:Wake()
            phys:ApplyForceCenter(ply:GetForward() * 50 + Vector(0, 0, 20))
        end
        
        return true
    end

    net.Receive("GiveWeaponToHands", function(len, ply)
        if not IsValid(ply) then return end
        local index = net.ReadUInt(16)
        if not ply.Inventory or not ply.Inventory[index] then 
            McINV.ForceUpdate(ply)
            McINV.RefreshUI(ply)
            McINV.ForceInventoryPage(ply)
            McINV.CloseUI(ply)
            return 
        end
        
        local item = ply.Inventory[index]
        if McINV.picktime_class[item.class] then
            local wep = ply:Give(item.class)
            if IsValid(wep) then
                if item.clip1 and item.clip1 > 0 then wep:SetClip1(item.clip1) end
                if item.clip2 and item.clip2 > 0 then wep:SetClip2(item.clip2) end
                if item.gunstate then wep.gunstate = item.gunstate end
                if item.att_saved then 
                    wep.att_saved = item.att_saved
                    if wep.BuildCustomizedGun then
                        timer.Simple(0.1, function()
                            if IsValid(wep) then
                                wep:BuildCustomizedGun()
                            end
                        end)
                    end
                end
                if item.modifications then
                    wep.modifications = item.modifications
                    if wep.ApplyModifications then
                        timer.Simple(0.1, function()
                            if IsValid(wep) then
                                wep:ApplyModifications()
                            end
                        end)
                    end
                end
                if item.ammo_type then
                    wep.AmmoType = item.ammo_type
                end
                if item.primary_ammo and item.primary_ammo > 0 then
                    wep:SetClip1(item.primary_ammo)
                end
                if item.secondary_ammo and item.secondary_ammo > 0 then
                    wep:SetClip2(item.secondary_ammo)
                end
                if item.weapon_data then
                    for k, v in pairs(item.weapon_data) do
                        wep[k] = v
                    end
                end
                ply:SelectWeapon(item.class)
            end
            table.remove(ply.Inventory, index)
            McINV.ForceUpdate(ply)
            McINV.RefreshUI(ply)
            McINV.ForceInventoryPage(ply)
            McINV.CloseUI(ply)
        else
            McINV.ForceUpdate(ply)
            McINV.RefreshUI(ply)
            McINV.ForceInventoryPage(ply)
            McINV.CloseUI(ply)
        end
    end)

    hook.Add("PlayerButtonDown", "McINV.PKMPickup", function(ply, button)
        if not IsValid(ply) then return end
        if button ~= MOUSE_RIGHT then return end
        if ply.ChoosingChar then return end
        if not IsFirstTimePredicted() then return end
        
        local tr = ply:GetEyeTrace()
        local ent = tr.Entity
        
        if not IsValid(ent) then return end
        if ply:GetPos():DistToSqr(ent:GetPos()) > 22500 then return end
        
        local itemClass = nil
        local weaponData = nil
        local pickupTime = 0.8
        
        if ent:IsWeapon() then
            local class = ent:GetClass()
            if McINV.picktime_class[class] then
                itemClass = class
                weaponData = {
                    clip1 = ent:Clip1(),
                    clip2 = ent:Clip2(),
                    gunstate = ent.gunstate or 1,
                    att_saved = ent.att_saved,
                    modifications = ent.modifications,
                    ammo_type = ent.AmmoType,
                    primary_ammo = ent:Clip1(),
                    secondary_ammo = ent:Clip2(),
                    weapon_data = {},
                }
                for k, v in pairs(ent) do
                    if type(v) ~= "function" and type(v) ~= "table" and k ~= "att_saved" and k ~= "modifications" and k ~= "AmmoType" then
                        weaponData.weapon_data[k] = v
                    end
                end
                pickupTime = math.max(0.5, McINV.picktime_class[class] * 0.2)
            end
        end
        
        if ent:GetClass() == "ent_bandage" then
            itemClass = "bandage"
            pickupTime = 0.5
        end
        
        if ent.IsWeaponDrop and ent.WeaponClass then
            if McINV.picktime_class[ent.WeaponClass] then
                itemClass = ent.WeaponClass
                weaponData = {
                    clip1 = ent.WeaponClip or 0,
                    gunstate = ent.WeaponState or 1,
                    att_saved = ent.Attachments,
                    modifications = ent.Modifications,
                    ammo_type = ent.AmmoType,
                    primary_ammo = ent.WeaponClip or 0,
                    secondary_ammo = ent.WeaponClip2 or 0,
                    weapon_data = ent.WeaponData or {},
                }
                pickupTime = math.max(0.5, McINV.picktime_class[itemClass] * 0.2)
            end
        end
        
        if ent:GetClass() == "spawned_weapon" then
            local class = ent:Geteclass()
            if class and McINV.picktime_class[class] then
                itemClass = class
                weaponData = {
                    clip1 = ent.WeaponClip1 or 0,
                    clip2 = ent.WeaponClip2 or 0,
                    gunstate = ent.gunstate or 1,
                    att_saved = ent.att_saved or {},
                    modifications = ent.modifications or {},
                    ammo_type = ent.AmmoType or "",
                    primary_ammo = ent.WeaponClip1 or 0,
                    secondary_ammo = ent.WeaponClip2 or 0,
                    weapon_data = ent.weapon_data or {},
                }
                pickupTime = math.max(0.5, McINV.picktime_class[itemClass] * 0.2)
            end
        end
        
        if itemClass then
            if pickupProgress[ply] then return end
            
            pickupProgress[ply] = {
                ent = ent,
                class = itemClass,
                data = weaponData,
                time = pickupTime,
                start = CurTime(),
            }
            
            net.Start("PickupProgress")
                net.WriteFloat(pickupTime)
                net.WriteFloat(0)
            net.Send(ply)
        end
    end)

    hook.Add("Think", "McINV.PickupProgressThink", function()
        local toRemove = {}
        
        for ply, data in pairs(pickupProgress) do
            if not IsValid(ply) then
                toRemove[ply] = true
            elseif not IsValid(data.ent) then
                toRemove[ply] = true
                net.Start("PickupProgress")
                    net.WriteFloat(0)
                    net.WriteFloat(0)
                net.Send(ply)
            elseif ply:GetPos():DistToSqr(data.ent:GetPos()) > 22500 then
                toRemove[ply] = true
                net.Start("PickupProgress")
                    net.WriteFloat(0)
                    net.WriteFloat(0)
                net.Send(ply)
            else
                local progress = (CurTime() - data.start) / data.time
                progress = math.min(progress, 1)
                
                net.Start("PickupProgress")
                    net.WriteFloat(data.time)
                    net.WriteFloat(progress)
                net.Send(ply)
                
                if progress >= 1 then
                    if IsValid(data.ent) then
                        if McINV.AddItem(ply, data.class, data.data) then
                            ply:EmitSound("items/ammo_pickup.wav")
                            data.ent:Remove()
                            McINV.ForceUpdate(ply)
                            McINV.RefreshUI(ply)
                            McINV.ForceInventoryPage(ply)
                        end
                    end
                    toRemove[ply] = true
                    net.Start("PickupProgress")
                        net.WriteFloat(0)
                        net.WriteFloat(0)
                    net.Send(ply)
                end
            end
        end
        
        for ply in pairs(toRemove) do
            pickupProgress[ply] = nil
        end
    end)

    hook.Add("PlayerUse", "McINV.UnpackWeaponBag", function(ply, ent)
        if not IsValid(ply) or not IsValid(ent) then return end
        if not ent.NY_IsWeaponBag then return end
        if unpackProgress[ply] then return end
        
        local unpackTime = ent.NY_IsHeavy and 3.0 or 1.5
        
        unpackProgress[ply] = {
            ent = ent,
            time = unpackTime,
            start = CurTime()
        }
        
        net.Start("UnpackBagProgress")
            net.WriteFloat(unpackTime)
            net.WriteFloat(0)
        net.Send(ply)
        
        return false
    end)

    hook.Add("Think", "McINV.UnpackBagThink", function()
        local toRemove = {}
        
        for ply, data in pairs(unpackProgress) do
            if not IsValid(ply) then
                toRemove[ply] = true
            elseif not IsValid(data.ent) then
                toRemove[ply] = true
                net.Start("UnpackBagProgress")
                    net.WriteFloat(0)
                    net.WriteFloat(0)
                net.Send(ply)
            elseif ply:GetPos():DistToSqr(data.ent:GetPos()) > 22500 then
                toRemove[ply] = true
                net.Start("UnpackBagProgress")
                    net.WriteFloat(0)
                    net.WriteFloat(0)
                net.Send(ply)
            else
                local progress = (CurTime() - data.start) / data.time
                progress = math.min(progress, 1)
                
                net.Start("UnpackBagProgress")
                    net.WriteFloat(data.time)
                    net.WriteFloat(progress)
                net.Send(ply)
                
                if progress >= 1 then
                    if IsValid(data.ent) then
                        local class = data.ent.NY_BagWeapon
                        local clip1 = data.ent.NY_BagClip1 or 0
                        local clip2 = data.ent.NY_BagClip2 or 0
                        local gunstate = data.ent.NY_BagGunstate or 1
                        local att_saved = data.ent.NY_BagAttachments or {}
                        local mods = data.ent.NY_BagMods or {}
                        local ammoType = data.ent.NY_BagAmmoType or ""
                        local primaryAmmo = data.ent.NY_BagPrimaryAmmo or 0
                        local secondaryAmmo = data.ent.NY_BagSecondaryAmmo or 0
                        local weaponData = data.ent.NY_BagWeaponData or {}
                        
                        local spawnPos = data.ent:GetPos()
                        data.ent:Remove()
                        
                        local weapon = ents.Create(class)
                        if IsValid(weapon) then
                            weapon:SetPos(spawnPos)
                            weapon:SetAngles(Angle(0, math.random(0, 360), 0))
                            weapon:Spawn()
                            weapon:Activate()
                            
                            if clip1 > 0 then weapon:SetClip1(clip1) end
                            if clip2 > 0 and weapon.SetClip2 then weapon:SetClip2(clip2) end
                            weapon.gunstate = gunstate
                            weapon.att_saved = att_saved
                            weapon.modifications = mods
                            weapon.AmmoType = ammoType
                            weapon.NY_IsWeaponDrop = true
                            weapon.NY_WeaponClass = class
                            weapon.NY_WeaponClip = clip1
                            weapon.NY_WeaponClip2 = clip2
                            weapon.NY_WeaponState = gunstate
                            weapon.NY_Attachments = att_saved
                            weapon.NY_Modifications = mods
                            weapon.NY_AmmoType = ammoType
                            weapon.NY_WeaponData = weaponData
                            
                            for k, v in pairs(weaponData) do
                                weapon[k] = v
                            end
                            
                            if weapon.BuildCustomizedGun then
                                timer.Simple(0.1, function()
                                    if IsValid(weapon) then
                                        weapon:BuildCustomizedGun()
                                    end
                                end)
                            end
                            
                            if weapon.ApplyModifications then
                                timer.Simple(0.15, function()
                                    if IsValid(weapon) then
                                        weapon:ApplyModifications()
                                    end
                                end)
                            end
                            
                            local phys = weapon:GetPhysicsObject()
                            if IsValid(phys) then
                                phys:Wake()
                            end
                        end
                    end
                    toRemove[ply] = true
                    net.Start("UnpackBagProgress")
                        net.WriteFloat(0)
                        net.WriteFloat(0)
                    net.Send(ply)
                end
            end
        end
        
        for ply in pairs(toRemove) do
            unpackProgress[ply] = nil
        end

    function McINV.RefreshUI()
        if IsValid(MSD.F4Menu) then
            if forceCloseUI then
                MSD.F4Menu:Remove()
                forceCloseUI = false
                needsUIUpdate = false
                return
            end
            
            if forceInventoryPage then
                MSD.F4Menu.currentPage = "inventory"
                forceInventoryPage = false
                
                for k, btn in pairs(MSD.F4Menu.buttons or {}) do
                    if btn and btn:IsValid() and btn.text == "Инвентарь" then
                        MSD.F4Menu.list.Deselect(btn)
                        btn.hovered = true
                    end
                end
            end
            
            if MSD.F4Menu.canvas then
                MSD.F4Menu.canvas:Clear()
                local currentPage = MSD.F4Menu.currentPage or "inventory"
                if currentPage == "inventory" then
                    McINV.CreateUI(MSD.F4Menu.canvas)
                end
            end
        end
        needsUIUpdate = false
    end

    hook.Add("Think", "McINV.UIRefreshThink", function()

            McINV.RefreshUI()
        end
    end)

    function McINV.InventoryItem(parent, item, index)
        local button = vgui.Create("DPanel")
        button:SetSize(100, 60)
        button.StaticScale = { w = 1, fixed_h = 60, minw = 250, minh = 60 }
        button.fade = 0
        
        local itemClass = item.class
        local itemName = McINV.GetWeaponName(itemClass)
        
        button.dname = itemName
        button.dweight = McINV.GetItemWeight(itemClass) .. " кг"
        button.isWeapon = McINV.picktime_class[itemClass] ~= nil
        button.gunstate = item.gunstate
        button.clip1 = item.clip1 or 0
        button.index = index
        button.modifications = item.modifications or {}
        button.att_saved = item.att_saved or {}

        local stateName, stateColor = McINV.GetGunState(item.gunstate)
        
        local modText = ""
        if button.isWeapon then
            local mods = {}
            if next(button.att_saved) then table.insert(mods, "AT") end
            if next(button.modifications) then table.insert(mods, "MOD") end
            if #mods > 0 then modText = " [" .. table.concat(mods, "+") .. "]" end
        end

        button.Paint = function(self, w, h)
            if (self.main and self.main.hover) or (self.sub and self.sub.hover) then
                self.fade = Lerp(FrameTime() * 7, self.fade, 1)
            else
                self.fade = Lerp(FrameTime() * 7, self.fade, 0)
            end
            
            draw.RoundedBox(0, 0, 0, w, h, MSD.Theme["d"])
            draw.RoundedBox(0, 0, 0, 5, h, MSD.Patern.MainColor["p"])
            
            local nameText = self.dname .. modText
            draw.SimpleText(nameText, "MSDFont.22", 55, 3, MMD.ColorAlpha(MMD.Text["d"], 255 - self.fade * 255), TEXT_ALIGN_LEFT)
            draw.DrawText(nameText, "MSDFont.22", 55, 3, MMD.ColorAlpha(MSD.Patern.MainColor["p"], self.fade * 255), TEXT_ALIGN_LEFT)
            
            draw.SimpleText(self.dweight, "MSDFont.16", 55, 28, MMD.Text["l"], TEXT_ALIGN_LEFT)
            
            if self.isWeapon then
                draw.SimpleText(stateName, "MSDFont.16", 55, 44, stateColor, TEXT_ALIGN_LEFT)
            end
            
            local iconMat = McINV.Icons[itemClass] or MSD.IconRender.Get("models/props_junk/CinderBlock01a.mdl")
            MSD.DrawTexturedRect(8, 8, 38, 38, iconMat or MSD.Icons48.box, color_white)
        end

        local dropBtn = MSD.IconButton(button, drop_icon, 5, 5, 46, nil, nil, function()
            if IsValid(LocalPlayer()) then
                net.Start("DropItemFromInventory")
                    net.WriteUInt(index, 16)
                net.SendToServer()
                if LocalPlayer().Inventory then
                    table.remove(LocalPlayer().Inventory, index)
                    needsUIUpdate = true
                    forceInventoryPage = true
                end
            end
        end)
        dropBtn:DockMargin(14, 14, 14, 14)
        dropBtn:DockPadding(14, 14, 14, 14)
        dropBtn:Dock(RIGHT)
        dropBtn:SetSize(32, 32)

        local useIcon = MSD.Icons48.check
        local useFunc = function()
            if IsValid(LocalPlayer()) then
                if button.isWeapon then
                    net.Start("GiveWeaponToHands")
                        net.WriteUInt(index, 16)
                    net.SendToServer()
                    if LocalPlayer().Inventory then
                        table.remove(LocalPlayer().Inventory, index)
                        needsUIUpdate = true
                        forceInventoryPage = true
                    end
                else
                    net.Start("UseItem")
                        net.WriteUInt(index, 16)
                    net.SendToServer()
                    if LocalPlayer().Inventory then
                        table.remove(LocalPlayer().Inventory, index)
                        needsUIUpdate = true
                        forceInventoryPage = true
                    end
                end
            end
        end
        
        if button.isWeapon then
            useIcon = gun_icon
        end
        
        local useBtn = MSD.IconButton(button, useIcon, 5, 5, 46, nil, nil, useFunc)
        useBtn:DockMargin(14, 14, 14, 14)
        useBtn:DockPadding(14, 14, 14, 14)
        useBtn:Dock(RIGHT)
        useBtn:SetSize(32, 32)

        parent:AddItem(button)
        return button
    end

    function McINV.CreateUI(parent)
        if not IsValid(parent) then return end
        if not IsValid(LocalPlayer()) then return end
        
        local list = vgui.Create("MSDPanelList", parent)
        list:SetSize(parent:GetWide(), parent:GetTall())
        list:SetPos(0, 0)
        list:EnableVerticalScrollbar()
        list:SetSpacing(2)
        list:SetPadding(0)

        local inv = LocalPlayer().Inventory or {}
        local curWeight = McINV.GetTotalWeight(LocalPlayer())
        local maxWeight = Config.MaxWeight or 8

        local header = vgui.Create("DPanel")
        header:SetSize(list:GetWide(), 50)
        header.Paint = function(self, w, h)
            draw.RoundedBox(MSD.Config.Rounded, 0, 0, w, h, MSD.Theme["l"])
            draw.SimpleText("Инвентарь", "MSDFont.24", 15, 5, MSD.Text.s, TEXT_ALIGN_LEFT)
            draw.SimpleText(curWeight .. " / " .. maxWeight .. " кг", "MSDFont.20", w - 15, 6, MSD.Text.n, TEXT_ALIGN_RIGHT)
            
            local pr = curWeight / maxWeight
            draw.RoundedBox(4, 15, 38, w - 30, 6, MSD.Patern.MainColor["9"])
            draw.RoundedBox(4, 15, 38, (w - 30) * math.min(pr, 1), 6, MSD.Patern.MainColor.p)
        end
        list:AddItem(header)

        for i, item in ipairs(inv) do
            McINV.InventoryItem(list, item, i)
        end

        return list
    end
